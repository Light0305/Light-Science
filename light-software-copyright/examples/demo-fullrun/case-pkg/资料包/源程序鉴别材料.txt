"""微统计算工具包：对数值序列计算描述统计并生成文本报表。"""
"""命令行入口：从标准输入或参数读数值序列，输出统计报表。"""
from __future__ import annotations
import sys
from .report import render_markdown, render_table
from .stats import summarize
def parse_values(argv):
    """把参数解析为浮点序列（逗号或空白分隔）。"""
    raw = " ".join(argv).replace(",", " ")
    return [float(x) for x in raw.split() if x]
def main(argv=None):
    """工具主入口：读数、汇总、按格式输出。"""
    args = list(sys.argv[1:] if argv is None else argv)
    fmt = "table"
    if args and args[0] == "--markdown":
        fmt = "markdown"
        args = args[1:]
    if not args:
        print("用法：tinystats [--markdown] 数值...", file=sys.stderr)
        return 2
    summary = summarize(parse_values(args))
    print(render_markdown(summary) if fmt == "markdown"
          else render_table(summary))
    return 0
if __name__ == "__main__":
    raise SystemExit(main())

def render_table(summary):
    """渲染统计汇总为文本表（异常值附行列出）。"""
    lines = ["指标        数值", "-" * 24]
    for key, label in LABELS:
        value = summary[key]
        shown = f"{value:.4f}" if isinstance(value, float) else str(value)
        lines.append(f"{label:<10}{shown}")
    outliers = summary.get("outliers") or []
    if outliers:
        lines.append(f"异常值下标：{outliers}")
    return "\n".join(lines)

"""描述统计计算：均值、中位数、方差、分位数与异常值标记。"""
from __future__ import annotations
import math
def mean(values):
    """计算算术平均值。"""
    if not values:
        raise ValueError("空序列无法计算均值")
    return sum(values) / len(values)
def median(values):
    """计算中位数（偶数个取中间两数均值）。"""
    ordered = sorted(values)
    n = len(ordered)
    mid = n // 2
    if n % 2:
        return ordered[mid]
    return (ordered[mid - 1] + ordered[mid]) / 2
def variance(values, sample=True):
    """计算方差；sample 为真时用样本方差（n-1）。"""
    n = len(values)
    if n < 2:
        return 0.0
    m = mean(values)
    total = sum((v - m) ** 2 for v in values)
    return total / (n - 1) if sample else total / n
def quantile(values, q):
    """线性插值分位数，q 取 0 到 1。"""
    ordered = sorted(values)
    if not ordered:
        raise ValueError("空序列无法计算分位数")
    pos = q * (len(ordered) - 1)
    lo = math.floor(pos)
    hi = math.ceil(pos)
    if lo == hi:
        return ordered[int(pos)]
    return ordered[lo] * (hi - pos) + ordered[hi] * (pos - lo)
def flag_outliers(values, k=1.5):
    """按 IQR 规则标记异常值下标。"""
    if len(values) < 4:
        return []
    q1 = quantile(values, 0.25)
    q3 = quantile(values, 0.75)
    iqr = q3 - q1
    lo, hi = q1 - k * iqr, q3 + k * iqr
    return [i for i, v in enumerate(values) if v < lo or v > hi]
